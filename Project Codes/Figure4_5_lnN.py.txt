import numpy as np
import matplotlib.pyplot as plt

# -----------------------------
# Natural Logarithm of Remaining Nuclei vs Time
# -----------------------------

# Parameters
N0 = 1000          # Initial number of nuclei
lam = 0.1          # Decay constant (s^-1)
t_max = 50         # Maximum time (s)

# Time array
time = np.linspace(0, t_max, 500)

# Analytical solution
N = N0 * np.exp(-lam * time)

# Natural logarithm of remaining nuclei
lnN = np.log(N)

# Plot
plt.figure(figsize=(7,5))
plt.plot(time, lnN, linewidth=2)

plt.xlabel("Time (s)", fontsize=12)
plt.ylabel("ln(N)", fontsize=12)

plt.grid(True, linestyle='--', alpha=0.7)

# Save figure
plt.savefig("lnN_vs_Time.png", dpi=300, bbox_inches="tight")

plt.show()
