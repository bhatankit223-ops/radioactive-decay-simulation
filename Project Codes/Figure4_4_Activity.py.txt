import numpy as np
import matplotlib.pyplot as plt

# -----------------------------
# Activity as a Function of Time
# -----------------------------

# Parameters
N0 = 1000          # Initial number of nuclei
lam = 0.1          # Decay constant (s^-1)
t_max = 50         # Maximum time (s)

# Time array
time = np.linspace(0, t_max, 500)

# Number of remaining nuclei
N = N0 * np.exp(-lam * time)

# Activity: A = λN
Activity = lam * N

# Plot
plt.figure(figsize=(7,5))
plt.plot(time, Activity, color='darkgreen', linewidth=2)

plt.xlabel("Time (s)", fontsize=12)
plt.ylabel("Activity A(t)", fontsize=12)

plt.grid(True, linestyle='--', alpha=0.7)

# Save figure
plt.savefig("Activity_vs_Time.png", dpi=300, bbox_inches="tight")

plt.show()
