import numpy as np
import matplotlib.pyplot as plt

# -------------------------------------
# Histogram of Decay Times
# -------------------------------------

# Parameters
N0 = 1000          # Initial number of nuclei
lam = 0.1          # Decay constant (s^-1)

# Set random seed for reproducibility
np.random.seed(42)

# Generate random decay times
decay_times = np.random.exponential(scale=1/lam, size=N0)

# Plot histogram
plt.figure(figsize=(7,5))
plt.hist(decay_times,
         bins=25,
         color='steelblue',
         edgecolor='black',
         alpha=0.8)

plt.xlabel("Decay Time (s)", fontsize=12)
plt.ylabel("Frequency", fontsize=12)

plt.grid(True, linestyle='--', alpha=0.7)

# Save figure
plt.savefig("Histogram_Decay_Times.png", dpi=300, bbox_inches="tight")

plt.show()
