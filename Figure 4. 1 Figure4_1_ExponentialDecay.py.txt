import numpy as np
import matplotlib.pyplot as plt

# Parameters
N0 = 1000          # Initial number of nuclei
lam = 0.1          # Decay constant (s^-1)
t_max = 50         # Maximum time (s)

# Time array
t = np.linspace(0, t_max, 500)

# Radioactive decay equation
N = N0 * np.exp(-lam * t)

# Plot
plt.figure(figsize=(7, 5))
plt.plot(t, N, color='blue', linewidth=2)

# Labels and title
plt.title("Figure 4.1: Exponential Radioactive Decay Curve", fontsize=14)
plt.xlabel("Time (s)", fontsize=12)
plt.ylabel("Number of Undecayed Nuclei, N(t)", fontsize=12)

# Grid
plt.grid(True, linestyle='--', alpha=0.7)

# Save figure (300 dpi for thesis)
plt.savefig("Figure4_1_Exponential_Decay.png", dpi=300, bbox_inches="tight")

# Show plot
plt.show()
